153enum LockRequestStatus {

161// 锁请求结构
162struct LockRequest {
163 int32_t transaction_id;
164 LockType mode;
165 int32_t page_id;
166 std::chrono::steady_clock::time_point request_time;
167 std::chrono::milliseconds timeout;
168 LockRequestStatus status;
169 std::function<void(LockRequestStatus)> callback; // 异步回调
170
171 LockRequest(int32_t tid, LockType m, int32_t pid,
172 std::chrono::milliseconds t = std::chrono::milliseconds(5000))
173 : transaction_id(tid), mode(m), page_id(pid),
174 request_time(std::chrono::steady_clock::now()),
175 timeout(t), status(WAITING) {}
176};

178// 锁持有信息
179struct LockHolder {
180 int32_t transaction_id;
181 LockType mode;
182 std::chrono::steady_clock::time_point acquire_time;
183 int32_t reference_count; // 引用计数，支持重入
184
185 LockHolder(int32_t tid, LockType m)

187 acquire_time(std::chrono::steady_clock::now()),
188 reference_count(1) {}
189};
