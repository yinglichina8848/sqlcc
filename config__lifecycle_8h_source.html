51enum class ConfigLifecycleState {
52 UNINITIALIZED, // The manager has not been initialized.
53 INITIALIZING, // The manager is currently loading the initial configuration.
54 READY, // The manager is initialized and ready to serve configuration snapshots.
55 UPDATING, // The manager is in the process of applying a new configuration snapshot.
56 SHUTTING_DOWN, // The manager is releasing resources.
57 SHUTDOWN // The manager is fully shut down.
58};

142 bool Initialize(const std::string& config_path = "") {
143 std::unique_lock<std::shared_mutex> lock(state_mutex_);
144
145 if (state_.load() != ConfigLifecycleState::UNINITIALIZED) {
146 return false;
147 }
148
149 try {
150 state_.store(ConfigLifecycleState::INITIALIZING);
151
152 if (!config_path.empty()) {
153 current_config_path_ = config_path;
154 }
155
156 // Execute user-defined initialization hook.
157 if (on_initialize_) {
158 on_initialize_();
159 }
160
161 // Create a default empty snapshot to ensure a valid state.
162 auto default_snapshot = ConfigSnapshotFactory::CreateEmptySnapshot(
163 GenerateVersionId("init_"), "Default configuration snapshot");
164
165 if (!snapshot_manager_.AddSnapshot(default_snapshot)) {
166 throw ConfigLifecycleException("Failed to add default snapshot", ConfigLifecycleState::INITIALIZING);
167 }
168
169 state_.store(ConfigLifecycleState::READY);
170 return true;
171
172 } catch (const std::exception& e) {
173 state_.store(ConfigLifecycleState::UNINITIALIZED);
174 ++error_count_;
175 throw ConfigLifecycleException(
176 std::string("Initialization failed: ") + e.what(),
177 ConfigLifecycleState::UNINITIALIZED);
178 }
179 }
180
187 bool Shutdown() {
188 std::unique_lock<std::shared_mutex> lock(state_mutex_);
189
190 if (state_.load() == ConfigLifecycleState::SHUTDOWN ||
191 state_.load() == ConfigLifecycleState::SHUTTING_DOWN) {
192 return true;
193 }
194
195 try {
196 state_.store(ConfigLifecycleState::SHUTTING_DOWN);
197
198 // Execute user-defined shutdown hook.
199 if (on_shutdown_) {
200 on_shutdown_();
201 }
202
203 // Clear all historical snapshots.
204 auto version_ids = snapshot_manager_.GetAllVersionIds();
205 for (const auto& version_id : version_ids) {
206 snapshot_manager_.RemoveSnapshot(version_id);
207 }
208
209 state_.store(ConfigLifecycleState::SHUTDOWN);
210 return true;
211
212 } catch (const std::exception& e) {
213 state_.store(ConfigLifecycleState::SHUTDOWN);
214 ++error_count_;
215 throw ConfigLifecycleException(
216 std::string("Shutdown failed: ") + e.what(),
217 ConfigLifecycleState::SHUTDOWN);
218 }
219 }

258 bool UpdateSnapshot(const ConfigSnapshot::SnapshotPtr& snapshot) {
259 if (!IsReady()) {
260 return false;
261 }
262
263 std::unique_lock<std::shared_mutex> lock(state_mutex_);
264
265 try {
266 state_.store(ConfigLifecycleState::UPDATING);
267
268 if (!snapshot) {
269 throw ConfigLifecycleException("Invalid snapshot", ConfigLifecycleState::UPDATING);
270 }
271
272 if (!snapshot_manager_.AddSnapshot(snapshot)) {
273 throw ConfigLifecycleException("Failed to add snapshot", ConfigLifecycleState::UPDATING);
274 }
275
276 last_update_time_ = std::chrono::steady_clock::now();
277 ++update_count_;
278
279 // Trigger the configuration change callback.
280 if (on_config_change_) {
281 on_config_change_(snapshot->GetMetadata().version_id);
282 }
283
284 state_.store(ConfigLifecycleState::READY);
285 return true;
286
287 } catch (const std::exception& e) {
288 state_.store(ConfigLifecycleState::READY); // Attempt to recover to a ready state.
289 ++error_count_;
290 throw ConfigLifecycleException(
291 std::string("Update failed: ") + e.what(),
292 ConfigLifecycleState::READY);
293 }
294 }

302 bool RollbackToVersion(const std::string& version_id) {

322 }

324 state_.store(ConfigLifecycleState::READY);
325 return true;

633 template<typename U>
634 static U ConvertConfigValue(const ConfigValue& value, const U& default_value) {
635 try {
636 if constexpr (std::is_same_v<U, bool>) {
637 if (std::holds_alternative<bool>(value)) {
638 return std::get<bool>(value);
639 } else if (std::holds_alternative<std::string>(value)) {
640 const auto& str = std::get<std::string>(value);
641 // Case-insensitive check for "true"
642 return (str.length() == 4 && (str[0] == 't' || str[0] == 'T') &&
643 (str[1] == 'r' || str[1] == 'R') &&
644 (str[2] == 'u' || str[2] == 'U') &&
645 (str[3] == 'e' || str[3] == 'E')) ||
646 (str == "1");
647 }
648 } else if constexpr (std::is_integral_v<U> && !std::is_same_v<U, bool>) {
649 if (std::holds_alternative<int>(value)) {
650 return static_cast<U>(std::get<int>(value));
651 } else if (std::holds_alternative<double>(value)) {
652 return static_cast<U>(std::get<double>(value));
653 }
654 } else if constexpr (std::is_floating_point_v<U>) {
655 if (std::holds_alternative<double>(value)) {
656 return static_cast<U>(std::get<double>(value));
657 } else if (std::holds_alternative<int>(value)) {
658 return static_cast<U>(std::get<int>(value));
659 }
660 } else if constexpr (std::is_same_v<U, std::string>) {
661 if (std::holds_alternative<std::string>(value)) {
662 return std::get<std::string>(value);
663 } else if (std::holds_alternative<bool>(value)) {
664 return std::get<bool>(value) ? "true" : "false";
665 } else if (std::holds_alternative<int>(value)) {
666 return std::to_string(std::get<int>(value));
667 } else if (std::holds_alternative<double>(value)) {
668 return std::to_string(std::get<double>(value));
669 }
670 }
671 } catch (const std::exception& e) {
672 // Conversion failed, fall through to return default value.
673 }
674
675 return default_value;
676 }
677
681 template<typename U>
682 static ConfigValue ConvertToConfigValue(const U& value) {
683 if constexpr (std::is_same_v<U, bool>) {
684 return value;
685 } else if constexpr (std::is_integral_v<U> && !std::is_same_v<U, bool>) {
686 return static_cast<int>(value);
687 } else if constexpr (std::is_floating_point_v<U>) {
688 return static_cast<double>(value);
689 } else if constexpr (std::is_convertible_v<U, std::string>) {
690 return std::string(value);
691 } else {
692 // If no direct conversion exists, default to converting to string.
693 // This requires that the type U is streamable.
694 std::ostringstream ss;
695 ss << value;
696 return ss.str();
697 }
698 }
