412 std::string CalculateChecksum() const {
413 // 简单的校验和计算，实际项目中可以使用更复杂的哈希算法
414 size_t hash = 0;
415 for (const auto& [key, value] : config_data_) {
416 std::hash<std::string> hasher;
417 hash ^= hasher(key) + 0x9e3779b9 + (hash << 6) + (hash >> 2);
418
419 // 对配置值也进行哈希
420 std::visit([&hash, &hasher](const auto& val) {
421 if constexpr (std::is_same_v<std::decay_t<decltype(val)>, std::string>) {
422 hash ^= hasher(val) + 0x9e3779b9 + (hash << 6) + (hash >> 2);
423 } else {
424 hash ^= std::hash<std::decay_t<decltype(val)>>{}(val) + 0x9e3779b9 + (hash << 6) + (hash >> 2);
425 }
426 }, value);
427 }
428
429 return std::to_string(hash);
430 }
431
436 bool ValidateIntegrity() const {
437 return CalculateChecksum() == metadata_.checksum;
438 }
439
444 SnapshotPtr Clone() const {
445 access_count_.fetch_add(1, std::memory_order_relaxed);
446
447 // 创建新的元数据，保持版本ID但更新时间
448 SnapshotMetadata new_metadata = metadata_;
449 new_metadata.create_time = std::chrono::system_clock::now();
450 new_metadata.parent_version = metadata_.version_id;
451
452 return std::make_shared<ConfigSnapshot>(config_data_, new_metadata);
453 }
454
460 bool Equals(const ConfigSnapshot& other) const {
461 return config_data_ == other.config_data_ &&
462 metadata_.version_id == other.metadata_.version_id;
463 }

480 static ConfigSnapshot::SnapshotPtr CreateSnapshot(
481 const std::unordered_map<std::string, ConfigValue>& config_data,
482 const std::string& version_id,
483 const std::string& description = "") {
484
485 ConfigSnapshot::SnapshotMetadata metadata;
486 metadata.version_id = version_id;
487 metadata.description = description;
488 metadata.create_time = std::chrono::system_clock::now();
489 metadata.config_count = config_data.size();
490
491 // 创建临时快照以计算校验和

493 metadata.checksum = temp_snapshot.CalculateChecksum();
494
495 return std::make_shared<ConfigSnapshot>(config_data, metadata);
496 }

505 const std::string& version_id,

520 static ConfigSnapshot::SnapshotPtr MergeSnapshots(
521 const ConfigSnapshot::SnapshotPtr& base_snapshot,
522 const ConfigSnapshot::SnapshotPtr& override_snapshot,
523 const std::string& new_version_id,
524 const std::string& description = "") {
525
526 if (!base_snapshot) {
527 return override_snapshot;
528 }
529
530 if (!override_snapshot) {
531 return base_snapshot;
532 }
533
534 // 获取基础配置数据
535 auto base_keys = base_snapshot->GetAllKeys();
536 std::unordered_map<std::string, ConfigValue> merged_data;
537
538 // 复制基础配置
539 for (const auto& key : base_keys) {
540 ConfigValue value;
541 if (base_snapshot->GetValue(key, value)) {
542 merged_data[key] = value;
543 }
544 }
545
546 // 应用覆盖配置
547 auto override_keys = override_snapshot->GetAllKeys();
548 for (const auto& key : override_keys) {
549 ConfigValue value;
550 if (override_snapshot->GetValue(key, value)) {
551 merged_data[key] = value; // 覆盖或新增
552 }
553 }
554
555 return CreateSnapshot(merged_data, new_version_id, description);
556 }

591 bool AddSnapshot(const ConfigSnapshot::SnapshotPtr& snapshot) {
592 if (!snapshot) {
593 return false;
594 }
595
596 std::unique_lock<std::shared_mutex> lock(snapshots_mutex_);
597
598 const auto& version_id = snapshot->GetMetadata().version_id;
599 snapshots_[version_id] = snapshot;
600
601 // 更新当前快照
602 current_snapshot_ = snapshot;
603 current_version_id_ = version_id;
604
605 // 更新版本历史
606 version_history_.push_back(version_id);
607 if (version_history_.size() > kMaxHistorySize) {
608 version_history_.erase(version_history_.begin());
609 }
610
611 return true;
612 }

653 bool RemoveSnapshot(const std::string& version_id) {
654 std::unique_lock<std::shared_mutex> lock(snapshots_mutex_);
655
656 // 不能删除当前版本
657 if (version_id == current_version_id_) {
658 return false;
659 }

662 }

669 size_t CleanupSnapshots(size_t keep_versions = 10) {
670 std::unique_lock<std::shared_mutex> lock(snapshots_mutex_);
671
672 if (snapshots_.size() <= keep_versions) {
673 return 0;
674 }
675
676 size_t removed_count = 0;
677 auto it = snapshots_.begin();
678
679 while (it != snapshots_.end() && snapshots_.size() > keep_versions) {
680 // 跳过当前版本
681 if (it->first == current_version_id_) {
682 ++it;
683 continue;
684 }
685
686 it = snapshots_.erase(it);
687 ++removed_count;
688 }
689
690 return removed_count;
691 }
