135 std::shared_ptr<ConnectionType> acquire(std::chrono::milliseconds timeout = std::chrono::milliseconds(5000)) {
136 std::unique_lock<std::mutex> lock(pool_mutex_);
137
138 // 首先尝试从空闲连接中获取
139 if (!idle_connections_.empty()) {
140 auto connection = idle_connections_.front();
141 idle_connections_.pop();
142
143 if (isConnectionValid(connection)) {
144 active_connections_.insert(connection);
145 stats_.active_connections++;
146 stats_.idle_connections--;
147 return connection;
148 } else {
149 // 连接无效，销毁并创建新连接
150 stats_.destroyed_connections++;
151 createConnection();
152 }
153 }
154
155 // 如果空闲连接不足，尝试创建新连接
156 if (active_connections_.size() + idle_connections_.size() < config_.max_size) {
157 createConnection();
158 if (!idle_connections_.empty()) {
159 auto connection = idle_connections_.front();
160 idle_connections_.pop();
161 active_connections_.insert(connection);
162 stats_.active_connections++;
163 return connection;
164 }
165 }
166
167 // 等待可用连接
168 stats_.waiting_threads++;
169 bool success = condition_.wait_for(lock, timeout, [this]() {
170 return !idle_connections_.empty() || !running_;
171 });
172 stats_.waiting_threads--;
173
174 if (!success || !running_) {
175 return nullptr;
176 }
177
178 if (!idle_connections_.empty()) {
179 auto connection = idle_connections_.front();
180 idle_connections_.pop();
181
182 if (isConnectionValid(connection)) {
183 active_connections_.insert(connection);
184 stats_.active_connections++;
185 stats_.idle_connections--;
186 return connection;
187 } else {
188 stats_.destroyed_connections++;
189 }
190 }
191
192 return nullptr;
193 }

199 void release(std::shared_ptr<ConnectionType> connection) {
200 if (!connection) return;
201
202 std::unique_lock<std::mutex> lock(pool_mutex_);
203
204 auto it = active_connections_.find(connection);
205 if (it != active_connections_.end()) {
206 active_connections_.erase(it);
207 stats_.active_connections--;
208
209 if (isConnectionValid(connection)) {
210 idle_connections_.push(connection);
211 stats_.idle_connections++;
212 } else {
213 stats_.destroyed_connections++;
214 }
215 }
216
217 condition_.notify_one();
218 }

298 void cleanupWorker() {
299 while (running_) {
300 std::this_thread::sleep_for(config_.health_check_interval);
301
302 std::unique_lock<std::mutex> lock(pool_mutex_);
303 if (!running_) break;
304
305 // 清理超时的空闲连接
306 auto now = std::chrono::steady_clock::now();
307 while (!idle_connections_.empty()) {
308 auto connection = idle_connections_.front();
309 // 简化实现：假设所有连接都是健康的，实际应记录连接创建时间
310 break;
311 }
312
313 // 确保最小连接数
314 size_t current_size = active_connections_.size() + idle_connections_.size();
315 if (current_size < config_.min_size && current_size < config_.max_size) {
316 size_t to_create = std::min(config_.min_size - current_size, config_.max_size - current_size);
317 for (size_t i = 0; i < to_create; ++i) {
318 lock.unlock();
319 createConnection();
320 lock.lock();
321 }
322 }
323 }
324 }
