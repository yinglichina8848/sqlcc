# 系统数据库组件的CMakeLists.txt
# 用于编译拆分后的系统数据库相关文件

# 设置源文件列表
set(SYSTEM_DATABASE_SOURCES
    system_schema_manager.cpp
    system_data_initializer.cpp
    system_permission_manager.cpp
    system_metadata_manager.cpp
    system_database_new.cpp
)

# 设置头文件列表
set(SYSTEM_DATABASE_HEADERS
    system_schema_manager.h
    system_data_initializer.h
    system_permission_manager.h
    system_metadata_manager.h
    system_database_new.h
    system_data_structures.h
)

# 创建系统数据库组件库
add_library(system_database ${SYSTEM_DATABASE_SOURCES} ${SYSTEM_DATABASE_HEADERS})

# 设置包含目录
target_include_directories(system_database PUBLIC 
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/src/core
)

# 链接依赖库
target_link_libraries(system_database 
    database_manager  # 假设这是数据库管理器的库名
)

# 设置编译选项
target_compile_options(system_database PRIVATE
    -Wall
    -Wextra
    -Wpedantic
)

# 设置C++标准
set_property(TARGET system_database PROPERTY CXX_STANDARD 17)
set_property(TARGET system_database PROPERTY CXX_STANDARD_REQUIRED ON)

# 安装规则
install(TARGETS system_database
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
)

install(FILES ${SYSTEM_DATABASE_HEADERS}
    DESTINATION include/sqlcc/sql_executor
)
