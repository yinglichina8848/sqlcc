279struct Value {
280 enum class Type { INT, DOUBLE, STRING } type;
281 union {
282 int64_t int_val;
283 double double_val;
284 };
285 std::string str_val;
286
287 Value() : type(Type::INT), int_val(0) {}
288 explicit Value(int64_t v) : type(Type::INT), int_val(v) {}
289 explicit Value(double v) : type(Type::DOUBLE), double_val(v) {}
290 explicit Value(const std::string& s) : type(Type::STRING), str_val(s) {}
291
292 // 重载==操作符，用于比较两个Value对象是否相等
293 bool operator==(const Value& other) const {
294 if (type != other.type) {
295 return false;
296 }
297 switch (type) {
298 case Type::INT:
299 return int_val == other.int_val;
300 case Type::DOUBLE:
301 return double_val == other.double_val;
302 case Type::STRING:
303 return str_val == other.str_val;
304 default:
305 return false;
306 }
307 }
308
309 // 重载!=操作符，用于比较两个Value对象是否不相等
310 bool operator!=(const Value& other) const {
311 return !(*this == other);
312 }
313
314 // 重载<操作符，用于比较两个Value对象
315 bool operator<(const Value& other) const {
316 if (type != other.type) {
317 return static_cast<int>(type) < static_cast<int>(other.type);
318 }
319 switch (type) {
320 case Type::INT:
321 return int_val < other.int_val;
322 case Type::DOUBLE:
323 return double_val < other.double_val;
324 case Type::STRING:
325 return str_val < other.str_val;
326 default:
327 return false;
328 }
329 }
330
331 // 重载>操作符，用于比较两个Value对象
332 bool operator>(const Value& other) const {
333 return other < *this;
334 }
335
336 // 转换为字符串
337 std::string toString() const {
338 switch (type) {
339 case Type::INT:
340 return std::to_string(int_val);
341 case Type::DOUBLE:
342 return std::to_string(double_val);
343 case Type::STRING:
344 return str_val;
345 default:
346 return "";
347 }
348 }
349
350 // 比较方法（用于排序）
351 int compare(const Value& other) const {
352 if (type != other.type) {
353 return static_cast<int>(type) - static_cast<int>(other.type);
354 }
355 switch (type) {
356 case Type::INT:
357 return (int_val < other.int_val) ? -1 : (int_val > other.int_val) ? 1 : 0;
358 case Type::DOUBLE:
359 return (double_val < other.double_val) ? -1 : (double_val > other.double_val) ? 1 : 0;
360 case Type::STRING:
361 return str_val.compare(other.str_val);
362 default:
363 return 0;
364 }
365 }
