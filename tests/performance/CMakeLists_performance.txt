# 性能测试框架配置
cmake_minimum_required(VERSION 3.10)
project(SQLCC_Performance_Tests)

# 包含目录
include_directories(${CMAKE_SOURCE_DIR}/src)
include_directories(${CMAKE_SOURCE_DIR}/src/parser)
include_directories(${CMAKE_SOURCE_DIR}/src/executor)

# 性能测试源文件
file(GLOB PERFORMANCE_TEST_FILES 
    "basic/*performance*.cc"
    "basic/*performance*.cpp"
    "advanced/*performance*.cc"
    "advanced/*performance*.cpp"
    "memory/*memory*.cc"
    "memory/*memory*.cpp"
    "concurrency/*concurrency*.cc"
    "concurrency/*concurrency*.cpp"
)

# 创建性能测试可执行文件
add_executable(performance_tests ${PERFORMANCE_TEST_FILES})
target_link_libraries(performance_tests 
    ${CMAKE_SOURCE_DIR}/libsqlcc.a
    pthread
    benchmark
    gtest
)

# 设置性能测试属性
set_target_properties(performance_tests PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests/performance"
)

# 添加性能测试
enable_testing()
add_test(NAME performance_test_basic COMMAND performance_tests)
