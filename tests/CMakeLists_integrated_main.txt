# SQLCC v1.1.1 集成测试框架主CMake文件
# 基于实际测试文件分布的智能配置

cmake_minimum_required(VERSION 3.10)
project(SQLCC_Integrated_Tests)

# 设置C++标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 启用测试
enable_testing()

# 包含目录
include_directories(
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_SOURCE_DIR}/include/core
    ${CMAKE_SOURCE_DIR}/include/network
    ${CMAKE_SOURCE_DIR}/include/sql
    ${CMAKE_SOURCE_DIR}/include/utils
)

# 查找必要的包
find_package(GTest REQUIRED)
find_package(Threads REQUIRED)

# 添加gtest宏
include_directories(${GTEST_INCLUDE_DIRS})

# 加载自动生成的测试配置
# 注意：这里我们手动包含关键测试以避免构建时间过长

# ===== HIGH 优先级核心测试 =====

# DCL测试 - 数据控制语言
set(DCL_TEST_SOURCES
    tests/test_revoke_persistence.cpp
    tests/test_grant_revoke.cpp
    tests/unit/basic/comprehensive_dcl_test.cpp
    tests/unit/basic/dcl_test.cpp
    tests/unit/basic/dcl_test_advanced.cpp
    tests/unit/basic/test_show_grants.cpp
    tests/unit/basic/permission_check_test.cpp
)

foreach(TEST_SOURCE ${DCL_TEST_SOURCES})
    if(EXISTS ${CMAKE_SOURCE_DIR}/${TEST_SOURCE})
        get_filename_component(TEST_NAME ${TEST_SOURCE} NAME_WE)
        add_executable(${TEST_NAME} ${TEST_SOURCE})
        target_link_libraries(${TEST_NAME}
            PRIVATE
            ${GTEST_LIBRARIES}
            GTest::Main
            Threads::Threads
            dl
            sqlcc_core_lib
            sqlcc_parser
            sqlcc_config_manager
            sqlcc_storage_engine
            sqlcc_transaction_manager
            sqlcc_executor
        )
        add_test(NAME ${TEST_NAME} COMMAND ${TEST_NAME})
        set_tests_properties(${TEST_NAME} PROPERTIES TIMEOUT 600)
    endif()
endforeach()
