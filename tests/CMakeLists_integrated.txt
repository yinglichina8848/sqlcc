add_test(NAME core_suite COMMAND ${CMAKE_CTEST_COMMAND} -R "(dcl|parser|executor|transaction|storage)_test")

# 快速测试套件
add_test(NAME quick_suite COMMAND ${CMAKE_CTEST_COMMAND} -R "(dcl|parser)_test" --output-on-failure)

# 完整测试套件
add_test(NAME full_suite COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure)

# 启用并行测试
set_property(TEST core_suite quick_suite full_suite PROPERTY PARALLEL TRUE)
