146template<class F, class... Args>
147auto ThreadPool::submit(F&& f, Args&&... args) -> std::future<typename std::invoke_result<F, Args...>::type> {
148 using return_type = typename std::invoke_result<F, Args...>::type;
149
150 // 将函数绑定为 packaged_task，以便能够通过 future 获取结果
151 auto task = std::make_shared<std::packaged_task<return_type()>>(
152 std::bind(std::forward<F>(f), std::forward<Args>(args)...)
153 );
154
155 std::future<return_type> result = task->get_future();
156
157 {
158 std::unique_lock<std::mutex> lock(queue_mutex_);
159 if (stop_) {
160 throw std::runtime_error("ThreadPool is stopped - cannot submit new task");
161 }
162
163 // 封装为 void 任务存入队列
164 tasks_.emplace([task]() { (*task)(); });
165 total_task_count_++;
166 }
167
168 // 唤醒一个正在等待任务的工作线程
169 condition_.notify_one();
170 return result;
171}
