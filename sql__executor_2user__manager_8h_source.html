63struct Permission {

65 std::string database;
66 std::string table;
67 std::string privilege;

71// 权限矩阵键结构（用于快速查找）
72struct PermissionKey {
73 std::string grantee;
74 std::string database;
75 std::string table;
76 std::string privilege;
77
78 bool operator==(const PermissionKey &other) const {
79 return grantee == other.grantee && database == other.database &&
80 table == other.table && privilege == other.privilege;
81 }
82};
83
84// 权限矩阵哈希函数
85struct PermissionKeyHash {
86 std::size_t operator()(const PermissionKey &key) const {
87 return std::hash<std::string>{}(key.grantee) ^
88 std::hash<std::string>{}(key.database) ^
89 std::hash<std::string>{}(key.table) ^
90 std::hash<std::string>{}(key.privilege);
91 }
92};
