163// 数据完整性约束类型
164enum IntegrityConstraintType {
165 NOT_NULL_CONSTRAINT = 1, // 非空约束
166 UNIQUE_CONSTRAINT = 2, // 唯一约束
167 PRIMARY_KEY_CONSTRAINT = 3, // 主键约束
168 FOREIGN_KEY_CONSTRAINT = 4, // 外键约束
169 CHECK_CONSTRAINT = 5, // 检查约束
170 DEFAULT_CONSTRAINT = 6 // 默认值约束

173// 约束验证结果
174enum ConstraintValidationResult {
175 CONSTRAINT_VALID = 0, // 约束验证通过
176 CONSTRAINT_VIOLATED = 1, // 约束违反
177 CONSTRAINT_NOT_FOUND = 2, // 约束不存在
178 CONSTRAINT_INVALID = 3, // 约束无效
179 REFERENCE_ERROR = 4 // 引用错误
180};

182// 约束规则定义
183struct ConstraintRule {
184 std::string constraint_name;
185 IntegrityConstraintType constraint_type;
186 std::string table_name;
187 std::vector<std::string> column_names;
188 std::string expression; // 检查约束表达式
189 std::string referenced_table; // 外键引用表
190 std::vector<std::string> referenced_columns; // 外键引用列
191 std::string default_value; // 默认值
192 bool enabled = true; // 约束是否启用
193 std::function<bool(const std::vector<std::string>&)> custom_validator; // 自定义验证函数
194
195 ConstraintRule(const std::string& name, IntegrityConstraintType type, const std::string& table)
196 : constraint_name(name), constraint_type(type), table_name(table) {}
197};
