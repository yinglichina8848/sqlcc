57class ConnectionStateMachine {
58public:
59 // 构造函数 - 增强异常安全
60 ConnectionStateMachine()
61 : current_state_(ConnectionState::DISCONNECTED),
62 target_state_(ConnectionState::DISCONNECTED),
63 is_transitioning_(false),
64 last_error_(""),
65 last_state_change_time_(std::chrono::steady_clock::now()) {}
66
67 // 析构函数 - 增强异常安全
68 ~ConnectionStateMachine() = default;
69
70 // 获取当前状态 - 线程安全
71 ConnectionState get_current_state() const {
72 std::lock_guard<std::mutex> lock(state_mutex_);
73 return current_state_;
74 }
75
76 // 检查是否可以进行指定转换 - 增强验证
77 bool can_transition(ConnectionEvent event) const {
78 std::lock_guard<std::mutex> lock(state_mutex_);
79 return is_valid_transition(current_state_, event);
80 }
81
82 // 执行状态转换 - 强保证异常安全
83 bool transition(ConnectionEvent event) {
84 std::lock_guard<std::mutex> lock(state_mutex_);
85
86 if (is_transitioning_) {
87 throw std::runtime_error("State transition already in progress");
88 }
89
90 ConnectionState new_state = get_next_state(current_state_, event);
91 if (new_state == current_state_) {
92 return false; // 无效转换
93 }
94
95 // 执行状态转换
96 is_transitioning_ = true;
97 ConnectionState old_state = current_state_;
98 current_state_ = new_state;
99 target_state_ = new_state;
100 last_state_change_time_ = std::chrono::steady_clock::now();
101 last_error_ = "";
102
103 is_transitioning_ = false;
104
105 // 通知状态变化
106 state_changed_cv_.notify_all();
107
108 return true;
109 }
110
111 // 设置目标状态 - 线程安全
112 void set_target_state(ConnectionState target_state) {
113 std::lock_guard<std::mutex> lock(state_mutex_);
114 target_state_ = target_state;
115 }
116
117 // 获取目标状态 - 线程安全
118 ConnectionState get_target_state() const {
119 std::lock_guard<std::mutex> lock(state_mutex_);
120 return target_state_;
121 }
122
123 // 等待状态变化 - 增强异常安全
124 bool wait_for_state_change(ConnectionState expected_state,
125 std::chrono::milliseconds timeout) {
126 std::unique_lock<std::mutex> lock(state_mutex_);
127 return state_changed_cv_.wait_for(lock, timeout,
128 [this, expected_state]() {
129 return current_state_ == expected_state && !is_transitioning_;
130 });
131 }
132
133 // 设置错误状态 - 线程安全
134 void set_error(const std::string& error_message) {
135 std::lock_guard<std::mutex> lock(state_mutex_);
136 current_state_ = ConnectionState::ERROR;
137 last_error_ = error_message;
138 last_state_change_time_ = std::chrono::steady_clock::now();
139 state_changed_cv_.notify_all();
140 }
141
142 // 获取最后错误 - 线程安全
143 std::string get_last_error() const {
144 std::lock_guard<std::mutex> lock(state_mutex_);
145 return last_error_;
146 }
147
148 // 检查是否正在转换 - 线程安全
149 bool is_transitioning() const {
150 std::lock_guard<std::mutex> lock(state_mutex_);
151 return is_transitioning_;
152 }
153
154 // 获取状态持续时间 - 线程安全
155 std::chrono::milliseconds get_state_duration() const {
156 std::lock_guard<std::mutex> lock(state_mutex_);
157 auto now = std::chrono::steady_clock::now();
158 return std::chrono::duration_cast<std::chrono::milliseconds>(
159 now - last_state_change_time_);
160 }
161
162 // 重置状态机 - 增强异常安全
163 void reset() {
164 std::lock_guard<std::mutex> lock(state_mutex_);
165 if (is_transitioning_) {
166 throw std::runtime_error("Cannot reset during state transition");
167 }
168 current_state_ = ConnectionState::DISCONNECTED;
169 target_state_ = ConnectionState::DISCONNECTED;
170 last_error_ = "";
171 last_state_change_time_ = std::chrono::steady_clock::now();
172 state_changed_cv_.notify_all();
173 }

176 // 状态转换验证 - 增强安全性
177 bool is_valid_transition(ConnectionState current, ConnectionEvent event) const {
178 switch (current) {
179 case ConnectionState::DISCONNECTED:
180 return event == ConnectionEvent::CONNECT_REQUEST ||
181 event == ConnectionEvent::ERROR;
182
183 case ConnectionState::CONNECTING:
184 return event == ConnectionEvent::CONNECT_SUCCESS ||
185 event == ConnectionEvent::CONNECT_FAILURE ||
186 event == ConnectionEvent::TIMEOUT ||
187 event == ConnectionEvent::DISCONNECT_REQUEST ||
188 event == ConnectionEvent::ERROR;
189
190 case ConnectionState::CONNECTED:
191 return event == ConnectionEvent::AUTHENTICATE_REQUEST ||
192 event == ConnectionEvent::DISCONNECT_REQUEST ||
193 event == ConnectionEvent::ERROR;
194
195 case ConnectionState::AUTHENTICATING:
196 return event == ConnectionEvent::AUTHENTICATE_SUCCESS ||
197 event == ConnectionEvent::AUTHENTICATE_FAILURE ||
198 event == ConnectionEvent::TIMEOUT ||
199 event == ConnectionEvent::DISCONNECT_REQUEST ||
200 event == ConnectionEvent::ERROR;
201
202 case ConnectionState::AUTHENTICATED:
203 return event == ConnectionEvent::DISCONNECT_REQUEST ||
204 event == ConnectionEvent::ERROR;
205
206 case ConnectionState::DISCONNECTING:
207 return event == ConnectionEvent::DISCONNECT_SUCCESS ||
208 event == ConnectionEvent::ERROR;
209
210 case ConnectionState::ERROR:
211 return event == ConnectionEvent::RECONNECT ||
212 event == ConnectionEvent::DISCONNECT_REQUEST;
213
214 case ConnectionState::RECONNECTING:
215 return event == ConnectionEvent::CONNECT_SUCCESS ||
216 event == ConnectionEvent::CONNECT_FAILURE ||
217 event == ConnectionEvent::TIMEOUT ||
218 event == ConnectionEvent::ERROR;
219
220 default:
221 return false;
222 }
223 }

225 // 获取下一个状态 - 增强验证
226 ConnectionState get_next_state(ConnectionState current, ConnectionEvent event) const {
227 if (!is_valid_transition(current, event)) {
228 throw std::invalid_argument("Invalid state transition: " +
229 std::to_string(static_cast<int>(current)) +
230 " -> " + std::to_string(static_cast<int>(event)));
231 }
232
233 switch (current) {
234 case ConnectionState::DISCONNECTED:
235 if (event == ConnectionEvent::CONNECT_REQUEST) {
236 return ConnectionState::CONNECTING;
237 }
238 break;
239
240 case ConnectionState::CONNECTING:
241 if (event == ConnectionEvent::CONNECT_SUCCESS) {
242 return ConnectionState::CONNECTED;
243 } else if (event == ConnectionEvent::CONNECT_FAILURE) {
244 return ConnectionState::ERROR;
245 } else if (event == ConnectionEvent::DISCONNECT_REQUEST) {
246 return ConnectionState::DISCONNECTED;
247 }
248 break;
249
250 case ConnectionState::CONNECTED:
251 if (event == ConnectionEvent::AUTHENTICATE_REQUEST) {
252 return ConnectionState::AUTHENTICATING;
253 } else if (event == ConnectionEvent::DISCONNECT_REQUEST) {
254 return ConnectionState::DISCONNECTING;
255 }
256 break;
257
258 case ConnectionState::AUTHENTICATING:
259 if (event == ConnectionEvent::AUTHENTICATE_SUCCESS) {
260 return ConnectionState::AUTHENTICATED;
261 } else if (event == ConnectionEvent::AUTHENTICATE_FAILURE) {
262 return ConnectionState::ERROR;
263 } else if (event == ConnectionEvent::DISCONNECT_REQUEST) {
264 return ConnectionState::DISCONNECTING;
265 }
266 break;
267
268 case ConnectionState::AUTHENTICATED:
269 if (event == ConnectionEvent::DISCONNECT_REQUEST) {
270 return ConnectionState::DISCONNECTING;
271 }
272 break;
273
274 case ConnectionState::DISCONNECTING:
275 if (event == ConnectionEvent::DISCONNECT_SUCCESS) {
276 return ConnectionState::DISCONNECTED;
277 }
278 break;
279
280 case ConnectionState::ERROR:
281 if (event == ConnectionEvent::RECONNECT) {
282 return ConnectionState::RECONNECTING;
283 } else if (event == ConnectionEvent::DISCONNECT_REQUEST) {
284 return ConnectionState::DISCONNECTED;
285 }
286 break;
287
288 case ConnectionState::RECONNECTING:
289 if (event == ConnectionEvent::CONNECT_SUCCESS) {
290 return ConnectionState::CONNECTED;
291 } else if (event == ConnectionEvent::CONNECT_FAILURE) {
292 return ConnectionState::ERROR;
293 }
294 break;
295 }
296
297 // 如果没有匹配的转换，保持当前状态
298 return current;
299 }

311class ConnectionTimeoutManager {
312public:
313 // 构造函数 - 增强异常安全
314 explicit ConnectionTimeoutManager(std::chrono::milliseconds connect_timeout =
315 std::chrono::seconds(30))
316 : connect_timeout_(connect_timeout),
317 authentication_timeout_(std::chrono::seconds(10)),
318 last_activity_time_(std::chrono::steady_clock::now()),
319 is_timeout_enabled_(true) {}
320
321 // 析构函数 - 增强异常安全
322 ~ConnectionTimeoutManager() = default;
323
324 // 设置连接超时 - 线程安全
325 void set_connect_timeout(std::chrono::milliseconds timeout) {
326 if (timeout <= std::chrono::milliseconds::zero()) {
327 throw std::invalid_argument("Connect timeout must be positive");
328 }
329 std::lock_guard<std::mutex> lock(timeout_mutex_);
330 connect_timeout_ = timeout;
331 }
332
333 // 设置认证超时 - 线程安全
334 void set_authentication_timeout(std::chrono::milliseconds timeout) {
335 if (timeout <= std::chrono::milliseconds::zero()) {
336 throw std::invalid_argument("Authentication timeout must be positive");
337 }
338 std::lock_guard<std::mutex> lock(timeout_mutex_);
339 authentication_timeout_ = timeout;
340 }
341
342 // 检查是否超时 - 增强验证
343 bool is_timeout(ConnectionState current_state) const {
344 std::lock_guard<std::mutex> lock(timeout_mutex_);
345
346 if (!is_timeout_enabled_) {
347 return false;
348 }
349
350 auto now = std::chrono::steady_clock::now();
351 auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(
352 now - last_activity_time_);
353
354 switch (current_state) {
355 case ConnectionState::CONNECTING:
356 return duration > connect_timeout_;
357 case ConnectionState::AUTHENTICATING:
358 return duration > authentication_timeout_;
359 default:
360 return false;
361 }
362 }
363
364 // 更新活动时间 - 线程安全
365 void update_activity_time() {
366 std::lock_guard<std::mutex> lock(timeout_mutex_);
367 last_activity_time_ = std::chrono::steady_clock::now();
368 }
369
370 // 启用/禁用超时 - 线程安全
371 void set_timeout_enabled(bool enabled) {
372 std::lock_guard<std::mutex> lock(timeout_mutex_);
373 is_timeout_enabled_ = enabled;
374 }
375
376 // 检查超时是否启用 - 线程安全
377 bool is_timeout_enabled() const {
378 std::lock_guard<std::mutex> lock(timeout_mutex_);
379 return is_timeout_enabled_;
380 }
381
382private:
383 std::chrono::milliseconds connect_timeout_;
384 std::chrono::milliseconds authentication_timeout_;
385 std::chrono::steady_clock::time_point last_activity_time_;
386 std::atomic<bool> is_timeout_enabled_;
387 mutable std::mutex timeout_mutex_;
388};
