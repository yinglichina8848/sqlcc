21struct PrefetchRequest {
22 int32_t page_id; // 页面ID
23 int32_t transaction_id; // 事务ID
24 std::chrono::steady_clock::time_point request_time; // 请求时间
25 int priority; // 优先级
26
27 PrefetchRequest(int32_t pid, int32_t tid, int p = 0)
28 : page_id(pid), transaction_id(tid), priority(p),
29 request_time(std::chrono::steady_clock::now()) {}
30
31 // 比较操作符，用于优先级队列
32 bool operator<(const PrefetchRequest &other) const {
33 // 优先级高的排在前面
34 if (priority != other.priority) {
35 return priority < other.priority;
36 }
37 // 优先级相同，请求时间早的排在前面
38 return request_time > other.request_time;
39 }
40};
