#### 2. Bazel构建系统
```bash
# Ubuntu
sudo apt install -y apt-transport-https curl gnupg
curl -fsSL https://bazel.build/bazel-release.pub.gpg | gpg --dearmor > bazel.gpg
sudo mv bazel.gpg /usr/share/keyrings/
echo "deb [arch=amd64 signed-by=/usr/share/keyrings/bazel.gpg] https://storage.googleapis.com/bazel-apt stable jdk1.8" | sudo tee /etc/apt/sources.list.d/bazel.list
sudo apt update && sudo apt install -y bazel

# CentOS
sudo yum install -y yum-utils
sudo yum-config-manager --add-repo https://copr.fedorainfracloud.org/coprs/vbatts/bazel/repo/epel-7/vbatts-bazel-epel-7.repo
sudo yum install -y bazel
```

#### 3. 依赖库安装
```bash
# OpenSSL
wget https://www.openssl.org/source/openssl-3.0.8.tar.gz
tar -xzf openssl-3.0.8.tar.gz && cd openssl-3.0.8
./config --prefix=/usr/local/openssl-3.0.8
make && sudo make install

# Google Test
git clone https://github.com/google/googletest.git -b v1.13.0
cd googletest && mkdir build && cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr/local/gtest-1.13.0 ..
make && sudo make install
```

## 🛠️ 多平台CI/CD配置

### GitHub Actions配置

#### Ubuntu环境
```yaml
name: Ubuntu CI

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  ubuntu-test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        compiler: [gcc-9, gcc-11, clang-10, clang-14]

    steps:
    - uses: actions/checkout@v3

    - name: Setup dependencies
      run: |
        sudo apt update
        sudo apt install -y ${{ matrix.compiler }} ninja-build libssl-dev libgtest-dev

    - name: Configure
      run: |
        mkdir build && cd build
        cmake -GNinja \
              -DCMAKE_CXX_COMPILER=${{ matrix.compiler == 'gcc-9' && 'g++-9' || matrix.compiler == 'gcc-11' && 'g++-11' || matrix.compiler == 'clang-10' && 'clang++-10' || 'clang++-14' }} \
              -DCMAKE_BUILD_TYPE=Release \
              ..

    - name: Build
      run: cd build && ninja

    - name: Test
      run: cd build && ctest --output-on-failure
```

#### macOS环境
```yaml
name: macOS CI

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  macos-test:
    runs-on: macos-latest

    steps:
    - uses: actions/checkout@v3

    - name: Setup dependencies
      run: |
        brew install llvm@14 ninja openssl googletest bazel

    - name: Configure
      run: |
        mkdir build && cd build
        cmake -GNinja \
              -DCMAKE_CXX_COMPILER=/usr/local/opt/llvm@14/bin/clang++ \
              -DCMAKE_BUILD_TYPE=Release \
              -DOPENSSL_ROOT_DIR=/usr/local/opt/openssl \
              ..

    - name: Build
      run: cd build && ninja

    - name: Test
      run: cd build && ctest --output-on-failure
```

#### Windows环境
```yaml
name: Windows CI

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  windows-test:
    runs-on: windows-latest

    steps:
    - uses: actions/checkout@v3

    - name: Setup MSVC
      uses: microsoft/setup-msbuild@v1.1

    - name: Configure
      run: |
        mkdir build
        cd build
        cmake -G "Visual Studio 17 2022" \
              -DCMAKE_BUILD_TYPE=Release \
              -Dgtest_force_shared_crt=ON \
              ..

    - name: Build
      run: |
        cd build
        msbuild SQLCC.sln /p:Configuration=Release /p:Platform=x64

    - name: Test
      run: |
        cd build
        ctest --output-on-failure -C Release
```

## 🔧 自动化测试脚本

### 环境检测脚本
```bash
#!/bin/bash
# detect_environment.sh

echo "=== SQLCC Environment Detection ==="
echo "Platform: $(uname -s)"
echo "Architecture: $(uname -m)"
echo "Kernel: $(uname -r)"

# 编译器检测
echo -e "\n=== Compiler Detection ==="
if command -v gcc &> /dev/null; then
    echo "GCC: $(gcc --version | head -n1)"
fi

if command -v clang &> /dev/null; then
    echo "Clang: $(clang --version | head -n1)"
fi

if command -v cl &> /dev/null; then
    echo "MSVC: $(cl 2>&1 | head -n1)"
fi

# 依赖库检测
echo -e "\n=== Library Detection ==="
if pkg-config --exists openssl; then
    echo "OpenSSL: $(pkg-config --modversion openssl)"
else
    echo "OpenSSL: NOT FOUND"
fi

if pkg-config --exists gtest; then
    echo "Google Test: $(pkg-config --modversion gtest)"
else
    echo "Google Test: NOT FOUND"
fi

# 构建工具检测
echo -e "\n=== Build Tools ==="
if command -v cmake &> /dev/null; then
    echo "CMake: $(cmake --version | head -n1)"
fi

if command -v bazel &> /dev/null; then
    echo "Bazel: $(bazel version | head -n1)"
fi

if command -v ninja &> /dev/null; then
    echo "Ninja: $(ninja --version)"
fi
```

### 统一构建脚本
```bash
#!/bin/bash
# build_all.sh

set -e

echo "=== SQLCC Multi-Platform Build ==="

# 检测环境
source scripts/detect_environment.sh

# 创建构建目录
BUILD_DIR="build_$(uname -s)_$(uname -m)"
mkdir -p "$BUILD_DIR"
cd "$BUILD_DIR"

# 根据平台选择构建方式
case "$(uname -s)" in
    Linux)
        echo "Building for Linux..."
        if command -v bazel &> /dev/null; then
            # 使用Bazel构建
            bazel build //...
        else
            # 使用CMake构建
            cmake -GNinja -DCMAKE_BUILD_TYPE=Release ..
            ninja
        fi
        ;;
    Darwin)
        echo "Building for macOS..."
        cmake -GNinja \
              -DCMAKE_CXX_COMPILER=/usr/local/opt/llvm/bin/clang++ \
              -DCMAKE_BUILD_TYPE=Release ..
        ninja
        ;;
    MINGW*|MSYS*)
        echo "Building for Windows..."
        cmake -G "Visual Studio 17 2022" \
              -DCMAKE_BUILD_TYPE=Release ..
        cmake --build . --config Release
        ;;
    *)
        echo "Unsupported platform: $(uname -s)"
        exit 1
        ;;
esac

echo "Build completed successfully!"
```

### 跨平台测试脚本
```bash
#!/bin/bash
# test_all.sh

set -e

echo "=== SQLCC Cross-Platform Testing ==="

# 运行单元测试
echo "Running unit tests..."
if [ -d "build" ]; then
    cd build
    if command -v ctest &> /dev/null; then
        ctest --output-on-failure
    elif command -v bazel &> /dev/null; then
        bazel test //...
    else
        echo "No test runner found"
        exit 1
    fi
else
    echo "Build directory not found. Run build_all.sh first."
    exit 1
fi

# 生成测试报告
echo "Generating test reports..."
if command -v lcov &> /dev/null; then
    lcov --capture --directory . --output-file coverage.info
    genhtml coverage.info --output-directory coverage_report
    echo "Coverage report generated: coverage_report/index.html"
fi

echo "All tests completed successfully!"
```
