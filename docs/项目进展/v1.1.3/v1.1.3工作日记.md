#### DMLExecutionStrategy::execute实现
```cpp
ExecutionResult DMLExecutionStrategy::execute(std::unique_ptr<sql_parser::Statement> stmt,
                                              ExecutionContext &context) {
  // 根据具体的语句类型执行
  if (auto insert_stmt = dynamic_cast<sql_parser::InsertStatement *>(stmt.get())) {
    return executeInsert(*insert_stmt, context);
  } else if (auto update_stmt = dynamic_cast<sql_parser::UpdateStatement *>(stmt.get())) {
    return executeUpdate(*update_stmt, context);
  } else if (auto delete_stmt = dynamic_cast<sql_parser::DeleteStatement *>(stmt.get())) {
    return executeDelete(*delete_stmt, context);
  } else if (auto select_stmt = dynamic_cast<sql_parser::SelectStatement *>(stmt.get())) {
    return executeSelect(*select_stmt, context);
  }

  return {false, "Unsupported DML statement type"};
}
```

### 任务：存储引擎接口内存安全重构

**问题描述：**
存储引擎接口存在内存安全隐患，主要问题包括：
- `StorageEngine::NewPage()` 和 `StorageEngine::FetchPage()` 方法返回裸指针 `Page*`
- `BufferPoolSharded::NewPage()` 和 `BufferPoolSharded::FetchPage()` 方法返回裸指针 `Page*`
- 内部使用智能指针管理但接口返回裸指针，存在生命周期管理不一致问题

**技术实现：**
1. **接口重构：**
   - 将 `Page* NewPage(int32_t* page_id)` 改为 `std::unique_ptr<Page> NewPage(int32_t* page_id)`
   - 将 `Page* FetchPage(int32_t page_id)` 改为 `std::unique_ptr<Page> FetchPage(int32_t page_id)`
   - 更新所有相关头文件和实现文件

2. **实现策略：**
   - 保持内部缓冲池继续使用 `std::shared_ptr<PageWrapper>` 管理页面生命周期
   - 接口返回时创建页面副本：`std::make_unique<Page>(*page_wrapper->page)`
   - 确保调用方获得独立的页面副本，避免共享状态

**代码变更：**
```
// 存储引擎头文件 (include/storage_engine.h)

- Page* FetchPage(int32_t page_id);

+ std::unique_ptr<Page> FetchPage(int32_t page_id);

// 缓冲池头文件 (include/storage/buffer_pool_sharded.h)
- Page* NewPage(int32_t* page_id);
- Page* FetchPage(int32_t page_id, bool exclusive = false);
+ std::unique_ptr<Page> NewPage(int32_t* page_id);
+ std::unique_ptr<Page> FetchPage(int32_t page_id, bool exclusive = false);

// 实现文件同步更新，返回页面副本而非裸指针
```

```
// ALTER语句AST节点定义
class AlterStatement : public Statement {
public:
  enum Target { DATABASE, TABLE };
  
  enum Action {
    ADD_COLUMN,
    DROP_COLUMN,
    MODIFY_COLUMN,
    RENAME_TABLE,
    ADD_INDEX,
    DROP_INDEX
  };

  // Getters
  Target getTarget() const;
  Action getAction() const;
  const std::string& getTableName() const;
  const std::string& getColumnName() const;
  const ColumnDefinition& getColumnDefinition() const;
  const std::string& getNewTableName() const;

  // Setters
  void setTableName(const std::string& name);
  void setAction(Action action);
  void setColumnName(const std::string& name);
  void setColumnDefinition(ColumnDefinition&& columnDef);
  void setNewTableName(const std::string& newName);
};

// 解析器实现示例
std::unique_ptr<Statement> ParserNew::parseAlterStatement() {
  consume(Token::KEYWORD_TABLE);
  
  std::string tableName = parseIdentifier();
  
  auto stmt = std::make_unique<AlterStatement>(AlterStatement::TABLE);
  stmt->setTableName(tableName);
  
  // 解析ALTER操作类型
  if (match(Token::KEYWORD_ADD)) {
    if (match(Token::KEYWORD_COLUMN)) {
      // ALTER TABLE table_name ADD COLUMN column_def
      auto columnDef = parseColumnDefinition();
      stmt->setAction(AlterStatement::ADD_COLUMN);
      stmt->setColumnDefinition(std::move(columnDef));
    }
  } else if (match(Token::KEYWORD_DROP)) {
    if (match(Token::KEYWORD_COLUMN)) {
      // ALTER TABLE table_name DROP COLUMN column_name
      std::string columnName = parseIdentifier();
      stmt->setAction(AlterStatement::DROP_COLUMN);
      stmt->setColumnName(columnName);
    }
  } else if (match(Token::KEYWORD_MODIFY)) {
    if (match(Token::KEYWORD_COLUMN)) {
      // ALTER TABLE table_name MODIFY COLUMN column_def
      auto columnDef = parseColumnDefinition();
      stmt->setAction(AlterStatement::MODIFY_COLUMN);
      stmt->setColumnDefinition(std::move(columnDef));
    }
  } else if (match(Token::KEYWORD_RENAME)) {
    if (match(Token::KEYWORD_TO)) {
      // ALTER TABLE table_name RENAME TO new_table_name
      std::string newTableName = parseIdentifier();
      stmt->setAction(AlterStatement::RENAME_TABLE);
      stmt->setNewTableName(newTableName);
    }
  }
  
  return stmt;
}
```

```
// 索引优化查询实现
std::vector<std::pair<int32_t, size_t>>
DMLExecutionStrategy::optimizeQueryWithIndex(
    const std::string &table_name, const sql_parser::WhereClause &where_clause,
    std::shared_ptr<StorageEngine> storage_engine, bool &used_index,
    std::string &index_info) {

  used_index = false;
  index_info = "全表扫描";

  if (where_clause.getColumnName().empty()) {
    TableStorageManager table_storage(storage_engine);
    return table_storage.ScanTable(table_name);
  }

  // 创建TableStorageManager实例
  TableStorageManager table_storage(storage_engine);
  
  // 检查是否有可用索引
  std::string column_name = where_clause.getColumnName();
  std::string op = where_clause.getOp();
  std::string value = where_clause.getValue();
  
  // 尝试获取表的索引
  BPlusTreeIndex* index = table_storage.GetIndex(table_name, column_name);
  
  // 如果有索引且操作符支持索引查找
  if (index && (op == "=" || op == ">" || op == ">=" || op == "<" || op == "<=")) {
    used_index = true;
    
    // 使用索引进行查找
    std::vector<IndexEntry> index_results;
    if (op == "=") {
      index_results = index->Search(value);
      index_info = "索引等式查询 (列: " + column_name + ", 值: " + value + ")";
    } else {
      // 根据操作符进行适当的范围搜索
      if (op == ">" || op == ">=") {
        // 大于或大于等于查询，使用范围搜索
        index_results = index->SearchRange(value, ""); // 空字符串表示无上限
        index_info = "索引范围查询 (列: " + column_name + ", 操作符: " + op + ", 值: " + value + ")";
      } else if (op == "<" || op == "<=") {

        index_results = index->SearchRange("", value); // 空字符串表示无下限

        index_results = index->Search(value);
        index_info = "索引等式查询 (列: " + column_name + ", 值: " + value + ")";
      }
    }
    
    // 将索引结果转换为位置向量
    std::vector<std::pair<int32_t, size_t>> locations;
    for (const auto& entry : index_results) {
      locations.emplace_back(entry.page_id, entry.offset);
    }
    
    return locations;
  }

  // ... 全表扫描实现
}
```

### 完成的任务

#### 1. BufferPool返回值逻辑重新设计
- **问题识别**: 原始实现中FetchPage返回`std::unique_ptr<Page>`，但实现中通过`std::make_unique<Page>(*page_table_[page_id])`创建拷贝，这导致缓冲池管理的shared_ptr和返回的unique_ptr拥有独立的对象，造成资源浪费和潜在的一致性问题
- **解决方案**: 将FetchPage和NewPage的返回类型改为`std::shared_ptr<Page>`，直接返回page_table_中已存储的智能指针，避免拷贝和多次销毁

#### 2. 核心接口重构
- **FetchPage方法**: 返回类型从`std::unique_ptr<Page>`改为`std::shared_ptr<Page>`，实现中直接返回`it->second`，确保调用者和缓冲池共享页面所有权
- **NewPage方法**: 返回类型从`std::unique_ptr<Page>`改为`std::shared_ptr<Page>`，实现中直接返回新创建的shared_ptr
- **BatchFetchPages方法**: 添加简化实现，支持批量获取页面
- **兼容性处理**: 为FetchPageShared和NewPageShared方法添加`[[deprecated]]`标记，引导用户使用新的接口

#### 3. 头文件接口更新
- **BufferPool类声明**: 更新FetchPage和NewPage方法的返回类型声明
- **废弃接口标记**: 为旧的Shared版本方法添加废弃警告

## 日期：2025年12月12日 - BufferPool智能指针改造完成

### 工作概述
今天的主要任务是重新设计BufferPool的智能指针逻辑，解决返回值的错误值问题，确保指针被正确管理而不被多次销毁。

### 完成的任务

#### 1. BufferPool返回值逻辑重新设计
- **问题识别**: 原始实现中FetchPage返回`std::unique_ptr<Page>`，但实现中通过`std::make_unique<Page>(*page_table_[page_id])`创建拷贝，这导致缓冲池管理的shared_ptr和返回的unique_ptr拥有独立的对象，造成资源浪费和潜在的一致性问题
- **解决方案**: 将FetchPage和NewPage的返回类型改为`std::shared_ptr<Page>`，直接返回page_table_中已存储的智能指针，避免拷贝和多次销毁

#### 2. 核心接口重构
- **FetchPage方法**: 返回类型从`std::unique_ptr<Page>`改为`std::shared_ptr<Page>`，实现中直接返回`it->second`，确保调用者和缓冲池共享页面所有权
- **NewPage方法**: 返回类型从`std::unique_ptr<Page>`改为`std::shared_ptr<Page>`，实现中直接返回新创建的shared_ptr
- **BatchFetchPages方法**: 添加简化实现，支持批量获取页面
- **兼容性处理**: 为FetchPageShared和NewPageShared方法添加`[[deprecated]]`标记，引导用户使用新的接口

#### 3. 头文件接口更新
- **BufferPool类声明**: 更新FetchPage和NewPage方法的返回类型声明
- **废弃接口标记**: 为旧的Shared版本方法添加废弃警告
- **文档注释更新**: 更新方法注释，说明新的智能指针语义

#### 4. 实现细节优化
- **页面存储**: page_table_继续使用`std::unordered_map<int32_t, std::shared_ptr<Page>>`，确保页面生命周期由智能指针管理
- **引用计数**: 通过shared_ptr的引用计数机制，确保页面在缓冲池和调用者之间正确共享
- **内存安全**: 消除了页面拷贝和多次释放的风险，提高了内存安全性
