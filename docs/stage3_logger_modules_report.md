```cpp

export module sqlcc.utils.logger;

// 标准库导入 (当前Clang 18不支持)
// import <string>;
// import <memory>;
// import <mutex>;

// 传统include作为替代

#include <memory>
#include <mutex>

export namespace sqlcc {
    class Logger { /* 导出接口 */ };
}

// src/utils/logger_module_impl.cpp (模块实现)
module sqlcc.utils.logger;
// 实现代码...
```

#### Clang 18 Modules限制识别
