### **自动化迁移脚本**

#### **migration_tool.py**
```python
#!/usr/bin/env python3
"""
SQLCC测试文件迁移工具
"""

import os
import shutil
from pathlib import Path

class TestMigrationTool:
    def __init__(self, root_dir: str):
        self.root_dir = Path(root_dir)
        self.migration_map = self.load_migration_map()

    def load_migration_map(self):
        """加载迁移映射配置"""
        return {
            'unit/basic/': 'level1_foundation/',
            'unit/storage/': 'level2_storage_engine/unit/',
            'integration/': 'level7_integration/core/',
            # ... 更多映射
        }

    def migrate_file(self, src_path: str, dst_path: str):
        """迁移单个文件"""
        src = self.root_dir / src_path
        dst = self.root_dir / dst_path

        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.move(str(src), str(dst))

        # 更新BUILD.bazel
        self.update_build_config(dst_path)

    def update_build_config(self, path: str):
        """更新BUILD配置"""
        build_file = Path(path).parent / 'BUILD.bazel'
        if build_file.exists():
            self.update_dependencies(build_file)

    def batch_migrate(self):
        """批量迁移"""
        for src_pattern, dst_dir in self.migration_map.items():
            src_files = list(self.root_dir.glob(f"{src_pattern}*.cpp"))
            for src_file in src_files:
                dst_file = dst_dir + src_file.name
                self.migrate_file(str(src_file), dst_file)
```
