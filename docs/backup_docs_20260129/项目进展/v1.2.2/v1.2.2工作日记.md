TEST(SqlExecutorTest, ComplexQueryExecution) {
    SqlExecutor executor(storage_engine, transaction_manager);
    
    std::string sql = "SELECT u.name, COUNT(o.id) as order_count "
                     "FROM users u "
                     "LEFT JOIN orders o ON u.id = o.user_id "
                     "WHERE u.status = 'active' "
                     "GROUP BY u.name "
                     "HAVING COUNT(o.id) > 0 "
                     "ORDER BY order_count DESC "
                     "LIMIT 10";
    
    auto result = executor.Execute(sql);
    ASSERT_TRUE(result.success);
    EXPECT_GT(result.rows.size(), 0);
    EXPECT_EQ(result.columns.size(), 2);
}
