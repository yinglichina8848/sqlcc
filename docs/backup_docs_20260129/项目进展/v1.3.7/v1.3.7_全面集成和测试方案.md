        threads.emplace_back([this, &success_count, i]() {

#### 4.3.4 性能测试框架设计

// tests/performance/performance_benchmark.h
template<typename Func>
double MeasureExecutionTime(Func&& func, int iterations = 1000) {
    auto start = std::chrono::high_resolution_clock::now();

    for (int i = 0; i < iterations; ++i) {
        func();
    }

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::microseconds>(end - start);
    return static_cast<double>(duration.count()) / iterations;
}

// 使用示例
TEST(PerformanceTest, BPlusTree_InsertPerformance) {
    BPlusTree tree;
    auto insert_time = MeasureExecutionTime([&]() {
        tree.Insert(GenerateRandomKey(), GenerateRandomValue());
    });

    // 性能断言
    EXPECT_LT(insert_time, 100.0); // 期望每次插入小于100微秒

    // 记录性能指标
    RecordBenchmark("bplus_tree_insert", insert_time);
}

### 5.1 CI流水线设计

```yaml
# .github/workflows/ci.yml
name: SQLCC CI Pipeline
on: [push, pull_request]

jobs:
  build_and_test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Bazel
        uses: bazelbuild/setup-bazelisk@v3

      - name: Build All
        run: bazel build //...

      - name: Run Unit Tests
        run: bazel test //tests/...

      - name: Coverage Analysis
        run: |
          bazel coverage //tests/... --combined_report=lcov
          genhtml bazel-out/_coverage/_coverage_report.dat \
            --output-directory coverage-html

      - name: Quality Gates
        run: |
          # 覆盖率门禁
          if [ $(lcov --summary coverage.info | grep "lines" | awk '{print $2}') -lt 80 ]; then
            echo "Coverage too low"
            exit 1
          fi

          # 编译警告检查
          if bazel build //... 2>&1 | grep -q "warning:"; then
            echo "Build warnings detected"
            exit 1
          fi
```
