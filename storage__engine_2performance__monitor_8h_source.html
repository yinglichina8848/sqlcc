37// 直方图桶配置
38struct HistogramBucket {
39 double upper_bound;
40 std::atomic<uint64_t> count;
41
42 HistogramBucket(double bound) : upper_bound(bound), count(0) {}
43
44 // 拷贝构造函数（需要处理原子变量的不可拷贝性）
45 HistogramBucket(const HistogramBucket &other)
46 : upper_bound(other.upper_bound), count(other.count.load()) {}
47
48 // 移动构造函数
49 HistogramBucket(HistogramBucket &&other) noexcept
50 : upper_bound(other.upper_bound), count(other.count.load()) {}
51
52 // 拷贝赋值操作符
53 HistogramBucket &operator=(const HistogramBucket &other) {
54 if (this != &other) {
55 upper_bound = other.upper_bound;
56 count.store(other.count.load());
57 }
58 return *this;
59 }
60
61 // 移动赋值操作符
62 HistogramBucket &operator=(HistogramBucket &&other) noexcept {
63 if (this != &other) {
64 upper_bound = other.upper_bound;
65 count.store(other.count.load());
66 }
67 return *this;
68 }
69};
